ખોટા દાવા યુક્રેનને શેડ્યૂલ પહેલાં સહાય મળી

345 કેપ્રેસન્ટ ડોનાલ્ડ ટ્રમ્પ કે જેમણે ગયા વર્ષે યુક્રેન માટે કરોડો ડોલર સુરક્ષા સહાયથી સ્થિર કર્યા હતા, તેઓએ દાવો કર્યો હતો કે શેડ્યૂલ ખોટા સમય પહેલાં તેઓ આ બધું મેળવ્યું હતું
ટ્રમ્પ દ્વારા નિર્દેશિત ફ્રીઝ ગયા ઉનાળાના લગભગ બે મહિના સુધી ચાલ્યો હતો અને પછીથી તમામ નાણાંનું વિતરણ કરવામાં આવ્યું ન હતું તે સમયસર કોંગ્રેસે વિસ્તરણ આપવું પડ્યું હતું કે સરકાર યુક્રેન માટે સહાય માટે કોઈપણ સંઘીય ભંડોળ ખર્ચ કરી શકે છે જે 30 સપ્ટેમ્બર સુધીમાં કરાર કરવામાં આવ્યો ન હતોજે કોંગ્રેસ દ્વારા સેટ કરેલી મૂળ સમયમર્યાદા હતી
હકીકતમાં લોસ એન્જલસ ટાઇમ્સે નવેમ્બરમાં અહેવાલ આપ્યો હતો કે સંરક્ષણ વિભાગ હજી યુક્રેન માટે તેની 250 મિલિયનથી વધુ સુરક્ષા સહાયતા અને ડિસેમ્બર સુધીમાં યુ.એસ. સાથી અખબાર માટે અનબિલિગેટેડ સંરક્ષણ ભંડોળમાં હજી 20 મિલિયન છે, ત્યાં હજી 20 મિલિયનથી વધુનું વિતરણ કર્યું નથીશોષિત
પેન્ટાગોનના એક અધિકારીએ અમને કહ્યું કે 23 જાન્યુઆરી સુધીમાં વિભાગે યુક્રેન માટેના 998 ભંડોળ ચલાવ્યું હતું અને બાકીની ફરજ માટે કામ કરી રહ્યું હતું
આ ભંડોળ પકડવાનો અને ત્યારબાદ યુક્રેનિયન રાષ્ટ્રપતિ વોલોડિમીર ઝેલેન્સ્કીને ડેમોક્રેટિક રાષ્ટ્રપતિ પદના ઉમેદવાર જ B બિડેન અને તેમના પુત્ર હન્ટરની તપાસ કરવા કહે છે, તે એક કારણ છે કે મોટાભાગના ગૃહના સભ્યોએ ડિસેમ્બરમાં ટ્રમ્પને મહાભિયોગ આપવા માટે મત આપ્યો હતો, જે રાષ્ટ્રપતિને દોષી ઠેરવશે અથવા નિર્દોષ જાહેર કરશે.આ અઠવાડિયે તેની મહાભિયોગની અજમાયશ
જ્યારે 21 જાન્યુઆરીએ સુનાવણી શરૂ થઈ ત્યારે ટ્રમ્પ એક દિવસ પછી વાર્ષિક વર્લ્ડ ઇકોનોમિક ફોરમ માટે ડેવોસ સ્વિટ્ઝર્લ in ન્ડમાં હતા, જ્યારે અન્ય વહીવટી અધિકારીઓ સાથેની એક પ્રેસ કોન્ફરન્સ દરમિયાન રાષ્ટ્રપતિએ યુક્રેન વિશે સમયસર તેની સહાય મેળવવાની દાવો કરી હતી
ટ્રમ્પ જાન્યુ 22 હવે હવે તેઓ યુક્રેનને તેમના પૈસા મળ્યા તે પહેલાં તેઓ તેમના બધા પૈસા મેળવે છે તેઓને તેમના બધા પૈસા મળ્યા તે મારા માટે ખૂબ મહત્વનું નથી, કેમ કે જર્મની શા માટે ચૂકવણી કરે છે કે યુકે કેમ ચૂકવણી કરે છે કેમ કે ફ્રાન્સ કેમ ચૂકવણી કરે છે કેમ કે યુરોપિયન રાષ્ટ્રો કેમ ચૂકવણી કરે છે કે કેમ ચૂકવણી કરે છેતે હંમેશાં સકર યુનાઇટેડ સ્ટેટ્સ
તે સાચું નથી કે જર્મની ફ્રાંસ યુનાઇટેડ કિંગડમ અને અન્ય યુરોપિયન દેશો યુક્રેનને સહાય પૂરી પાડતા નથી કારણ કે આપણે યુરોપિયન યુનિયન અને યુરોપિયન નાણાકીય સંસ્થાઓએ 2014 થી યુક્રેનને અનુદાન અને લોનમાં 164 અબજથી વધુનું યોગદાન આપ્યું છે, જ્યારે રશિયાએ ક્રિમીઆને જોડ્યો અને શરૂ કર્યોપૂર્વી યુક્રેનમાં ડોનબાસ ક્ષેત્રમાં સંઘર્ષ
અને યુક્રેનને શેડ્યૂલ પહેલાં યુ.એસ. તરફથી તેની સહાય મળી નથી
નાણાકીય વર્ષ 2019 માટે કોંગ્રેસે તાલીમ સાધનો અને તે રકમના યુક્રેન માટેના અન્ય સમર્થન સહિતના 391 મિલિયનની સુરક્ષા સહાયતા, 250 મિલિયનની રકમના યુક્રેન સુરક્ષા સહાય પહેલ માટે સંરક્ષણ વિભાગને ફાળવવામાં આવ્યા હતા અને વિદેશી સૈન્ય ધિરાણ માટે રાજ્ય વિભાગને 1415 મિલિયન ફાળવવામાં આવ્યા હતા.પ્રોગ્રામ 30 સપ્ટેમ્બર 2019 સુધીમાં તે વિભાગો દ્વારા ભંડોળનો ખર્ચ કરવામાં આવ્યો હતો
જૂનમાં સંરક્ષણ વિભાગે યુક્રેનને 250 મિલિયન સુરક્ષા સહાય પૂરી પાડવાની તેની યોજનાની ઘોષણા કરી હતી, જોકે જુલાઈમાં ટ્રમ્પે વ્હાઇટ હાઉસ Office ફિસ Management ફ મેનેજમેન્ટ અને બજેટને સહાય આપી હતી કે તેઓને ઓબીબી પણ રાજ્ય વિભાગની સહાયને અવરોધિત કરી હતી.
11 સપ્ટેમ્બરના રોજ લગભગ બે મહિના પછી તે ન હતું જ્યારે કોંગ્રેસ અને વહીવટી અધિકારીઓના સભ્યોના દબાણ હેઠળ વ્હાઇટ હાઉસ આ નાણાં જાહેર કર્યા
પરંતુ કારણ કે સંરક્ષણ વિભાગે વધુ 15 દિવસ રાહ જોવી જરૂરી હતી તે પહેલાં તે સપ્ટેમ્બરના અંત પહેલા બધા પૈસા ખર્ચવા માટે સક્ષમ ન હતું, જ્યારે ફેડરલ નાણાકીય વર્ષ સમાપ્ત થાય છે
કોંગ્રેસે સતત ફાળવણી બિલમાં જોગવાઈ ઉમેરવી પડી હતી, જે ટ્રમ્પે 27 સપ્ટેમ્બરના રોજ કાયદામાં સહી કરી હતી, જેથી નાણાકીય વર્ષ 2020 માં અનપેન્ટ ફંડનો ઉપયોગ કરવાની મંજૂરી આપી
ઓએમબીમાં નેશનલ સિક્યુરિટીના ડેપ્યુટી એસોસિએટ ડિરેક્ટર માર્ક સેન્ડીએ તેમના 16 નવેમ્બરના જુબાની પર જુબાની આપી હતી કે સંરક્ષણ વિભાગ દ્વારા આશરે 35 મિલિયન જેટલા અસંભવિત રહ્યા હતા, તે ભંડોળ તેમણે જણાવ્યું હતું કે જો કોંગ્રેસના પગથિયા માટે નહીં
પછી ત્રણ દિવસ પછી લોસ એન્જલસ ટાઇમ્સે અહેવાલ આપ્યો કે પેન્ટાગોનના દસ્તાવેજોની સમીક્ષા કરવામાં આવેલા પેન્ટાગોન દસ્તાવેજો અનુસાર હજી પણ તે ભંડોળ ખર્ચવામાં આવ્યું નથી અને 12 ડિસેમ્બરે પછીના મહિનામાં ટાઇમ્સે અહેવાલ આપ્યો છે કે લગભગ 20 મિલિયન સહાય યુક્રેન પહોંચી નથી
જે દર્શાવે છે કે રાષ્ટ્રપતિએ કહ્યું તેમ તમામ ભંડોળ શેડ્યૂલ પર આગળ વધ્યું ન હતું
